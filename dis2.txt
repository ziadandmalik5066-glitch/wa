def get_target_channel(guild: discord.Guild, channel_str: str) -> discord.TextChannel:
    channel_str = channel_str.strip()
    if channel_str.startswith("<#") and channel_str.endswith(">"):
        channel_id = channel_str[2:-1]
        if channel_id.isdigit():
            return guild.get_channel(int(channel_id))
    elif channel_str.isdigit():
        return guild.get_channel(int(channel_str))
    else:
        return discord.utils.get(guild.text_channels, name=channel_str)
    return None

class SayModal(discord.ui.Modal, title="Message Setup"):
    def __init__(self, user_id: int, trigger_message: discord.Message = None):
        super().__init__()
        self.user_id = user_id
        self.trigger_message = trigger_message

    target_channel_input = discord.ui.TextInput(label="Channel", required=True, placeholder="Channel iD")
    message_text = discord.ui.TextInput(label="Message", style=discord.TextStyle.paragraph, required=True, placeholder="Your Message")

    async def on_submit(self, interaction: discord.Interaction):
        if interaction.user.id != self.user_id:
            await interaction.response.send_message("**You Don't Have Permission** <a:error:1554589848520884396>", ephemeral=True)
            return

        target_channel = get_target_channel(interaction.guild, self.target_channel_input.value)

        if not target_channel:
            await interaction.response.send_message("**Can't Find The Channel** <a:error:1554589848520884396>", ephemeral=True)
            return

        try:
            await target_channel.send(self.message_text.value.strip())
            await interaction.response.send_message(f"**Done** <a:done:1554957071638859870> {target_channel.mention}!", ephemeral=True)
        except discord.HTTPException:
            await interaction.response.send_message("**Error Sending Message** <a:error:1554589848520884396>", ephemeral=True)

        if self.trigger_message:
            try:
                await self.trigger_message.delete()
            except discord.HTTPException:
                pass

class TriggerButtonView(discord.ui.View):
    def __init__(self, user_id: int, modal_type: str):
        super().__init__(timeout=None)
        self.user_id = user_id
        self.modal_type = modal_type
        self.message = None

        if self.modal_type == "say":
            say_button = discord.ui.Button(label="Setup", style=discord.ButtonStyle.secondary, emoji="<a:updatingyellow:1555622787635347469>")
            say_button.callback = self.open_say
            self.add_item(say_button)

        elif self.modal_type == "embed":
            embed_button = discord.ui.Button(label="Setup", style=discord.ButtonStyle.secondary, emoji="<a:updatingblue:1555622876713853001>")
            embed_button.callback = self.open_embed
            self.add_item(embed_button)

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        if interaction.user.id != self.user_id:
            await interaction.response.send_message("**You Don't Have Permission** <a:error:1554589848520884396>", ephemeral=True)
            return False
        return True

    async def open_say(self, interaction: discord.Interaction):
        await interaction.response.send_modal(SayModal(user_id=self.user_id, trigger_message=self.message or interaction.message))

    async def open_embed(self, interaction: discord.Interaction):
        await interaction.response.send_modal(NormalEmbedModal(user_id=self.user_id, trigger_message=self.message or interaction.message))
